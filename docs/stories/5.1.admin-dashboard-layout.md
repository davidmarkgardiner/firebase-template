### Admin Layout Component
```typescript
interface AdminLayoutProps {
  children: React.ReactNode
}

function AdminLayout({ children }: AdminLayoutProps) {
  const { user, isLoading } = useAdminAuth()
  const [sidebarOpen, setSidebarOpen] = useState(false)

  if (isLoading) {
    return <AdminLoadingSkeleton />
  }

  if (!user || !user.isAdmin) {
    redirect('/admin/login')
  }

  return (
    <div className="min-h-screen bg-gray-100">
      <AdminSidebar open={sidebarOpen} onClose={() => setSidebarOpen(false)} />
      <div className="lg:pl-64">
        <AdminHeader onMenuClick={() => setSidebarOpen(true)} />
        <main className="p-6">
          {children}
        </main>
      </div>
    </div>
  )
}
```

### Admin Authentication & RBAC
```typescript
interface AdminUser extends User {
  role: 'admin' | 'manager' | 'staff'
  permissions: Permission[]
  lastLogin: Date
}

interface Permission {
  action: string // 'read', 'write', 'delete'
  resource: string // 'orders', 'products', 'customers'
}

// Database schema for admin roles
CREATE TABLE admin_users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id),
  role VARCHAR(20) NOT NULL CHECK (role IN ('admin', 'manager', 'staff')),
  permissions JSONB DEFAULT '[]'::jsonb,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  last_login TIMESTAMPTZ
);

// Row Level Security for admin access
CREATE POLICY "Admin users can access admin tables" ON admin_users
  FOR ALL USING (
    EXISTS (
      SELECT 1 FROM admin_users au
      WHERE au.user_id = auth.uid()
    )
  );
```

### Activity Feed System
```typescript
interface ActivityEvent {
  id: string
  type: 'order_created' | 'product_updated' | 'user_registered' | 'payment_failed'
  description: string
  userId?: string
  metadata: Record<string, any>
  createdAt: Date
}

// Real-time activity updates using Supabase Realtime
function ActivityFeed() {
  const [activities, setActivities] = useState<ActivityEvent[]>([])

  useEffect(() => {
    const subscription = supabase
      .channel('admin-activities')
      .on('postgres_changes', {
        event: 'INSERT',
        schema: 'public',
        table: 'activity_log'
      }, (payload) => {
        setActivities(prev => [payload.new as ActivityEvent, ...prev.slice(0, 49)])
      })
      .subscribe()

    return () => subscription.unsubscribe()
  }, [])

  return (
    <div className="bg-white rounded-lg shadow-sm p-6">
      <h3 className="text-lg font-semibold mb-4">Recent Activity</h3>
      <div className="space-y-3">
        {activities.map(activity => (
          <ActivityItem key={activity.id} activity={activity} />
        ))}
      </div>
    </div>
  )
}
```

### Inventory Alerts System
```typescript
interface InventoryAlert {
  productId: string
  productName: string
  currentStock: number
  threshold: number
  severity: 'low' | 'critical' | 'out_of_stock'
  lastRestocked: Date
}

function InventoryAlerts() {
  const { data: alerts } = useSWR('/api/admin/inventory/alerts', fetchInventoryAlerts)

  const criticalAlerts = alerts?.filter(alert => alert.severity === 'critical') || []
  const lowStockAlerts = alerts?.filter(alert => alert.severity === 'low') || []

  return (
    <div className="bg-white rounded-lg shadow-sm p-6">
      <h3 className="text-lg font-semibold mb-4">Inventory Alerts</h3>
      {criticalAlerts.length > 0 && (
        <div className="mb-4">
          <h4 className="text-red-600 font-medium mb-2">Critical Alerts</h4>
          <div className="space-y-2">
            {criticalAlerts.map(alert => (
              <InventoryAlertItem key={alert.productId} alert={alert} />
            ))}
          </div>
        </div>
      )}
      {lowStockAlerts.length > 0 && (
        <div>
          <h4 className="text-yellow-600 font-medium mb-2">Low Stock</h4>
          <div className="space-y-2">
            {lowStockAlerts.map(alert => (
              <InventoryAlertItem key={alert.productId} alert={alert} />
            ))}
          </div>
        </div>
      )}
    </div>
  )
}
```
