### Product Data Structure
```typescript
interface Product {
  id: string
  name: string
  slug: string
  description: string
  shortDescription?: string
  status: 'active' | 'draft' | 'archived'
  featured: boolean

  // Coffee-specific fields
  originFarm: string
  altitude: number
  processingMethod: string
  tastingNotes: string[]
  roastLevel: 'light' | 'medium' | 'dark'
  harvestDate: Date
  story: string

  // SEO fields
  metaTitle?: string
  metaDescription?: string

  // Timestamps
  createdAt: Date
  updatedAt: Date

  // Relations
  variants: ProductVariant[]
  images: ProductImage[]
  categories: Category[]
}

interface ProductVariant {
  id: string
  productId: string
  sku: string
  format: 'whole_bean' | 'ground'
  grindType?: 'espresso' | 'filter' | 'french_press' | 'moka'
  weight: number // in grams
  price: number
  compareAtPrice?: number
  stockQuantity: number
  lowStockThreshold: number
  allowBackorder: boolean
  barcode?: string
}

interface ProductImage {
  id: string
  productId: string
  url: string
  altText: string
  sortOrder: number
  isMain: boolean
}
```

### Product Table Component
```typescript
interface ProductTableProps {
  products: Product[]
  loading: boolean
  onEdit: (product: Product) => void
  onDelete: (productId: string) => void
  onBulkAction: (action: string, productIds: string[]) => void
}

function ProductTable({ products, loading, onEdit, onDelete, onBulkAction }: ProductTableProps) {
  const [selectedProducts, setSelectedProducts] = useState<string[]>([])
  const [sortConfig, setSortConfig] = useState<{ key: string, direction: 'asc' | 'desc' }>()

  const columns = [
    {
      key: 'name',
      title: 'Product',
      render: (product: Product) => (
        <div className="flex items-center space-x-3">
          <img src={product.images[0]?.url} className="w-10 h-10 rounded-lg" />
          <div>
            <div className="font-medium">{product.name}</div>
            <div className="text-sm text-gray-500">{product.sku}</div>
          </div>
        </div>
      )
    },
    {
      key: 'status',
      title: 'Status',
      render: (product: Product) => <StatusBadge status={product.status} />
    },
    {
      key: 'stock',
      title: 'Stock',
      render: (product: Product) => (
        <StockIndicator
          current={product.variants.reduce((sum, v) => sum + v.stockQuantity, 0)}
          threshold={product.variants[0]?.lowStockThreshold || 10}
        />
      )
    },
    {
      key: 'price',
      title: 'Price',
      render: (product: Product) => {
        const minPrice = Math.min(...product.variants.map(v => v.price))
        const maxPrice = Math.max(...product.variants.map(v => v.price))
        return minPrice === maxPrice ? `€${minPrice}` : `€${minPrice} - €${maxPrice}`
      }
    }
  ]

  return (
    <DataTable
      columns={columns}
      data={products}
      loading={loading}
      selectable
      selectedRows={selectedProducts}
      onSelectionChange={setSelectedProducts}
      sortConfig={sortConfig}
      onSort={setSortConfig}
      actions={
        <BulkActions
          selectedCount={selectedProducts.length}
          onAction={(action) => onBulkAction(action, selectedProducts)}
        />
      }
    />
  )
}
```

### Product Form Component
```typescript
const productSchema = z.object({
  name: z.string().min(1, 'Product name is required'),
  slug: z.string().min(1, 'URL slug is required'),
  description: z.string().min(10, 'Description must be at least 10 characters'),
  originFarm: z.string().min(1, 'Origin farm is required'),
  altitude: z.number().min(0, 'Altitude must be positive'),
  processingMethod: z.string().min(1, 'Processing method is required'),
  tastingNotes: z.array(z.string()).min(1, 'At least one tasting note required'),
  roastLevel: z.enum(['light', 'medium', 'dark']),
  harvestDate: z.date(),
  status: z.enum(['active', 'draft', 'archived'])
})

function ProductForm({ product, onSave }: { product?: Product, onSave: (data: any) => void }) {
  const [currentStep, setCurrentStep] = useState(0)
  const [images, setImages] = useState<File[]>([])

  const form = useForm({
    resolver: zodResolver(productSchema),
    defaultValues: product || {
      status: 'draft',
      featured: false,
      tastingNotes: [],
      roastLevel: 'medium'
    }
  })

  const steps = [
    { title: 'Basic Information', component: BasicInfoStep },
    { title: 'Coffee Details', component: CoffeeDetailsStep },
    { title: 'Images', component: ImageUploadStep },
    { title: 'Variants', component: VariantsStep },
    { title: 'SEO', component: SEOStep }
  ]

  return (
    <div className="max-w-4xl mx-auto">
      <StepIndicator steps={steps} currentStep={currentStep} />

      <form onSubmit={form.handleSubmit(onSave)} className="mt-8">
        <div className="bg-white rounded-lg shadow-sm p-6">
          {React.createElement(steps[currentStep].component, {
            form,
            images,
            setImages
          })}

          <div className="flex justify-between mt-8">
            <Button
              type="button"
              variant="outline"
              onClick={() => setCurrentStep(Math.max(0, currentStep - 1))}
              disabled={currentStep === 0}
            >
              Previous
            </Button>

            {currentStep === steps.length - 1 ? (
              <Button type="submit">Save Product</Button>
            ) : (
              <Button
                type="button"
                onClick={() => setCurrentStep(Math.min(steps.length - 1, currentStep + 1))}
              >
                Next
              </Button>
            )}
          </div>
        </div>
      </form>
    </div>
  )
}
```

### Inventory Management System
```typescript
interface InventoryAdjustment {
  id: string
  productVariantId: string
  type: 'restock' | 'sale' | 'damage' | 'return' | 'correction'
  quantityChange: number
  reason: string
  notes?: string
  adjustedBy: string
  createdAt: Date
}

function InventoryManager() {
  const [adjustments, setAdjustments] = useState<InventoryAdjustment[]>([])
  const [showAdjustmentModal, setShowAdjustmentModal] = useState(false)
  const [selectedVariant, setSelectedVariant] = useState<ProductVariant | null>(null)

  async function handleStockAdjustment(adjustment: Omit<InventoryAdjustment, 'id' | 'createdAt'>) {
    // Update local stock
    const { error } = await supabase
      .from('product_variants')
      .update({
        stock_quantity: selectedVariant!.stockQuantity + adjustment.quantityChange
      })
      .eq('id', adjustment.productVariantId)

    // Log the adjustment
    await supabase
      .from('inventory_adjustments')
      .insert(adjustment)

    // Refresh data
    mutate('/api/admin/inventory')
  }

  return (
    <div className="space-y-6">
      <div className="flex justify-between items-center">
        <h1 className="text-2xl font-semibold">Inventory Management</h1>
        <Button onClick={() => setShowAdjustmentModal(true)}>
          Adjust Stock
        </Button>
      </div>

      <InventoryTable
        variants={variants}
        onAdjust={(variant) => {
          setSelectedVariant(variant)
          setShowAdjustmentModal(true)
        }}
      />

      <AdjustmentHistory adjustments={adjustments} />

      {showAdjustmentModal && (
        <StockAdjustmentModal
          variant={selectedVariant}
          onSave={handleStockAdjustment}
          onClose={() => {
            setShowAdjustmentModal(false)
            setSelectedVariant(null)
          }}
        />
      )}
    </div>
  )
}
```

### Image Management System
```typescript
function ImageGallery({ productId, images, onImagesChange }: ImageGalleryProps) {
  const [uploading, setUploading] = useState(false)

  async function handleImageUpload(files: File[]) {
    setUploading(true)

    try {
      const uploadPromises = files.map(async (file) => {
        const fileName = `products/${productId}/${Date.now()}-${file.name}`

        const { data, error } = await supabase.storage
          .from('product-images')
          .upload(fileName, file)

        if (error) throw error

        return {
          url: `${supabaseUrl}/storage/v1/object/public/product-images/${fileName}`,
          altText: '',
          sortOrder: images.length,
          isMain: images.length === 0
        }
      })

      const newImages = await Promise.all(uploadPromises)
      onImagesChange([...images, ...newImages])
    } catch (error) {
      toast.error('Failed to upload images')
    } finally {
      setUploading(false)
    }
  }

  return (
    <div className="space-y-4">
      <ImageUploadDropzone
        onFilesSelected={handleImageUpload}
        loading={uploading}
        accept="image/*"
        multiple
      />

      <DragDropContext onDragEnd={handleImageReorder}>
        <Droppable droppableId="images">
          {(provided) => (
            <div
              {...provided.droppableProps}
              ref={provided.innerRef}
              className="grid grid-cols-2 md:grid-cols-4 gap-4"
            >
              {images.map((image, index) => (
                <Draggable key={image.id} draggableId={image.id} index={index}>
                  {(provided) => (
                    <div
                      ref={provided.innerRef}
                      {...provided.draggableProps}
                      {...provided.dragHandleProps}
                      className="relative group"
                    >
                      <img
                        src={image.url}
                        alt={image.altText}
                        className="w-full h-32 object-cover rounded-lg"
                      />
                      <div className="absolute inset-0 bg-black bg-opacity-0 group-hover:bg-opacity-50 transition-opacity rounded-lg flex items-center justify-center">
                        <div className="opacity-0 group-hover:opacity-100 flex space-x-2">
                          <Button size="sm" variant="outline" onClick={() => editImage(image)}>
                            Edit
                          </Button>
                          <Button size="sm" variant="outline" onClick={() => deleteImage(image.id)}>
                            Delete
                          </Button>
                        </div>
                      </div>
                      {image.isMain && (
                        <div className="absolute top-2 left-2 bg-green-500 text-white text-xs px-2 py-1 rounded">
                          Main
                        </div>
                      )}
                    </div>
                  )}
                </Draggable>
              ))}
              {provided.placeholder}
            </div>
          )}
        </Droppable>
      </DragDropContext>
    </div>
  )
}
```

### Bulk Operations System
```typescript
interface BulkOperation {
  type: 'update_status' | 'update_price' | 'update_stock' | 'export' | 'import'
  data: any
  productIds?: string[]
}

function BulkOperations({ selectedProducts }: { selectedProducts: string[] }) {
  const [operation, setOperation] = useState<BulkOperation | null>(null)

  async function handleBulkStatusUpdate(status: string) {
    const { error } = await supabase
      .from('products')
      .update({ status })
      .in('id', selectedProducts)

    if (!error) {
      toast.success(`Updated ${selectedProducts.length} products`)
      mutate('/api/admin/products')
    }
  }

  async function handleCSVExport() {
    const products = await fetchProductsByIds(selectedProducts)
    const csv = generateCSV(products)
    downloadFile(csv, 'products.csv', 'text/csv')
  }

  async function handleCSVImport(file: File) {
    const csv = await file.text()
    const products = parseCSV(csv)

    // Validate products
    const validation = validateProductsCSV(products)
    if (validation.errors.length > 0) {
      toast.error(`Validation failed: ${validation.errors.join(', ')}`)
      return
    }

    // Import products
    const { error } = await supabase
      .from('products')
      .upsert(products)

    if (!error) {
      toast.success(`Imported ${products.length} products`)
      mutate('/api/admin/products')
    }
  }

  return (
    <div className="flex items-center space-x-2">
      <Select onValueChange={(value) => handleBulkStatusUpdate(value)}>
        <SelectTrigger>
          <SelectValue placeholder="Update Status" />
        </SelectTrigger>
        <SelectContent>
          <SelectItem value="active">Set Active</SelectItem>
          <SelectItem value="draft">Set Draft</SelectItem>
          <SelectItem value="archived">Archive</SelectItem>
        </SelectContent>
      </Select>

      <Button onClick={handleCSVExport}>Export CSV</Button>

      <FileUpload
        onFileSelected={handleCSVImport}
        accept=".csv"
        trigger={<Button variant="outline">Import CSV</Button>}
      />
    </div>
  )
}
```
