# Story 4.2: User Account Dashboard

## Status
Draft

## Story
**As a** registered Honduras Coffee customer,
**I want** a comprehensive account dashboard to manage my profile and orders,
**so that** I can track purchases, update preferences, and manage subscriptions easily.

1. Account dashboard overview with recent orders and quick actions
2. Profile management with personal information editing

4. Address book management for shipping and billing addresses
5. Subscription management for coffee deliveries
6. Payment methods management with saved cards
7. Preference settings for notifications and marketing
8. Download invoices and receipts (PDF)
9. Account security settings (password change, 2FA)
10. Mobile-optimized dashboard experience
11. Loading states and error handling
12. Data export functionality (GDPR compliance)

```typescript
interface Order {
```

### Subscription Management
```typescript
interface Subscription {
  id: string
  stripeSubscriptionId: string
  status: 'active' | 'paused' | 'cancelled'
  product: Product
  frequency: 'weekly' | 'biweekly' | 'monthly'
  nextDelivery: Date
  quantity: number
  price: number
}

async function updateSubscription(id: string, updates: Partial<Subscription>) {
  // Update local subscription
  const { error } = await supabase
    .from('subscriptions')
    .update(updates)
    .eq('id', id)

  // Update Stripe subscription
  await fetch('/api/stripe/subscriptions', {
    method: 'PATCH',
    body: JSON.stringify({ subscriptionId: id, ...updates })
  })
}
```

### Address Book Management
```typescript
interface Address {
  id: string
  type: 'shipping' | 'billing'
  firstName: string
  lastName: string
  company?: string
  address1: string
  address2?: string
  city: string
  postalCode: string
  country: string
  phone?: string
  isDefault: boolean
}

function AddressBook() {
  const [addresses, setAddresses] = useState<Address[]>([])
  const [editingAddress, setEditingAddress] = useState<Address | null>(null)

  async function saveAddress(address: Omit<Address, 'id'>) {
    const { data, error } = await supabase
      .from('user_addresses')
      .insert({
        ...address,
        user_id: user.id
      })
      .select()
      .single()

    if (!error) {
      setAddresses(prev => [...prev, data])
    }
  }

  return (
    <DashboardLayout title="Address Book">
      <AddressGrid addresses={addresses} onEdit={setEditingAddress} />
      {editingAddress && (
        <AddressModal
          address={editingAddress}
          onSave={saveAddress}
          onClose={() => setEditingAddress(null)}
        />
      )}
    </DashboardLayout>
  )
}
```

### Payment Methods Integration
```typescript
// Stripe Customer Portal integration
async function openCustomerPortal() {
  const response = await fetch('/api/stripe/customer-portal', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      customerId: user.stripeCustomerId,
      returnUrl: `${window.location.origin}/account/payments`
    })
  })

  const { url } = await response.json()
  window.location.href = url
}

function PaymentMethods() {
  const [paymentMethods, setPaymentMethods] = useState([])

        ))}
        <Button onClick={openCustomerPortal}>
          Manage Payment Methods
        </Button>
      </div>
    </DashboardLayout>
  )
}
```

### Invoice Generation
```typescript
import PDFKit from 'pdfkit'

async function generateInvoice(orderId: string): Promise<Buffer> {
  const order = await getOrderWithDetails(orderId)

  const doc = new PDFKit()

  // Company header
  doc.fontSize(20).text('Honduras Mountain Coffee', 50, 50)
  doc.fontSize(12).text('Invoice', 50, 100)

  // Order details
  doc.text(`Order #${order.orderNumber}`, 50, 130)
  doc.text(`Date: ${order.createdAt.toLocaleDateString()}`, 50, 150)

  // Items
  let yPosition = 200
  order.items.forEach(item => {
    doc.text(`${item.productName} x ${item.quantity}`, 50, yPosition)
    doc.text(`€${item.price}`, 400, yPosition)
    yPosition += 20
  })

  // Total
  doc.text(`Total: €${order.total}`, 50, yPosition + 20)

  return doc
}
```

```typescript
async function exportUserData(userId: string) {
  const userData = await Promise.all([
    getUserProfile(userId),
    getUserOrders(userId),
    getUserAddresses(userId),
    getUserSubscriptions(userId),
    getUserPreferences(userId)
  ])

  const dataPackage = {
    profile: userData[0],
    orders: userData[1],
    addresses: userData[2],
    subscriptions: userData[3],
    preferences: userData[4],
    exportedAt: new Date().toISOString()
  }

  return JSON.stringify(dataPackage, null, 2)
}
```
